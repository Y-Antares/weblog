// ================= 配置区域 =================
const sourceTags = ['#游戏记录', '#影视记录']; 
const sortOrder = 'desc'; 
// ===========================================

let pages = dv.pages(sourceTags.join(" or "))
    .filter(p => !p.file.path.includes("Template") && p.file.name !== "未命名" && p.file.name !== "game record");

function getImageUrl(imagePath) {
    if (!imagePath) return "";
    if (typeof imagePath === 'string' && imagePath.startsWith('http')) return imagePath;
    let filename = imagePath;
    if (imagePath.path) filename = imagePath.path; 
    const file = app.metadataCache.getFirstLinkpathDest(filename, "");
    if (file) return app.vault.getResourcePath(file);
    return ""; 
}

function formatDate(date) {
    if (!date) return '-';
    if (date.toFormat) return date.toFormat("yyyy-MM-dd");
    return String(date).substring(0, 10);
}

if (pages.length === 0) {
    dv.paragraph("请确保笔记包含相关标签。");
} else {
    pages = pages.sort(p => p["观影日期"] || p["通关日期"], sortOrder);

    const container = document.createElement("div");
    container.className = "media-gallery";
    dv.container.appendChild(container);

    pages.forEach((page, index) => {
        let isMovie = page.tags && (page.tags.includes("影视记录") || page.tags.includes("#影视记录"));
        let cardClass = isMovie ? "media-card movie-card" : "media-card game-card";
        let coverUrl = getImageUrl(page["封面"]);
        
        let displayTags = page.tags || [];
        if (typeof displayTags === 'string') displayTags = [displayTags];
        const tagsHtml = displayTags
            .filter(t => t !== '游戏记录' && t !== '影视记录' && t !== '#游戏记录' && t !== '#影视记录') 
            .map(t => `<span class="media-tag">${t}</span>`)
            .join('');

        let rank = page["序号"] || (index + 1);
        let score = page["评分"] || '-';
        let finishDate = formatDate(page["观影日期"] || page["通关日期"]);

        const card = document.createElement("div");
        card.className = cardClass;
        card.innerHTML = `
            <div class="media-card-cover" style="background-image: url('${coverUrl}');">
                <div class="media-rank">#${rank}</div>
                <div class="media-score-badge">${score}</div>
            </div>
            <div class="media-card-content">
                <div class="media-title-cn">${page["中文名"] || page.file.name}</div>
                <div class="media-title-en">${page["英文名"] || ''}</div>
                <div class="media-stars">${"★".repeat(Math.round((parseFloat(score)||0)/2))}${"☆".repeat(5-Math.round((parseFloat(score)||0)/2))}</div>
                <div class="media-comment">${page["评价"] || '暂无评价'}</div>
                <div class="media-tags">${tagsHtml}</div>
                <div class="media-meta-row">
                    <div class="media-meta-item"><span class="meta-icon">🕒</span><span>${String(page["时长"]||'-').replace('PT','').replace('H','h')}</span></div>
                    <div class="media-meta-item"><span class="meta-icon">📺</span><span>${page["平台"]||(isMovie?'影院':'PC')}</span></div>
                    <div class="media-meta-item"><span class="meta-icon">📅</span><span>${finishDate}</span></div>
                </div>
            </div>
        `;
        card.onclick = (e) => { e.preventDefault(); app.workspace.openLinkText(page.file.path, "", false); };
        container.appendChild(card);
    });
}
